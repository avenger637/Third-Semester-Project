## Nested class


Use collision handling (e.g., separate chaining), a load-factor threshold, and correct key equality. Average lookup is O(1); worst case is O(n).

```java
import java.util.Objects;
public class HashTable<K,V> {
    private static class Entry<K,V>{final K key;V value;Entry(K key,V value){this.key=key;this.value=value;}}
    private LinkedList<Entry<K,V>>[] buckets; private int capacity,size;
    @SuppressWarnings("unchecked") public HashTable(int capacity){if(capacity<1)throw new IllegalArgumentException("Capacity must be positive");this.capacity=capacity;buckets=new LinkedList[capacity];for(int i=0;i<capacity;i++)buckets[i]=new LinkedList<>();}
    public int hash(K key){Objects.requireNonNull(key,"Key required");return (key.hashCode()&Integer.MAX_VALUE)%capacity;}
    public void put(K key,V value){Objects.requireNonNull(key,"Key required");Objects.requireNonNull(value,"Value required");int h=hash(key);LinkedList<Entry<K,V>> b=buckets[h];for(int i=0;i<b.size();i++){Entry<K,V> e=b.get(i);if(Objects.equals(e.key,key)){e.value=value;return;}}b.addLast(new Entry<>(key,value));size++;if((double)size/capacity>=0.75)resize();}
    public V get(K key){Objects.requireNonNull(key,"Key required");LinkedList<Entry<K,V>> b=buckets[hash(key)];for(int i=0;i<b.size();i++){Entry<K,V> e=b.get(i);if(Objects.equals(e.key,key))return e.value;}return null;}
    public boolean containsKey(K key){Objects.requireNonNull(key,"Key required");LinkedList<Entry<K,V>> b=buckets[hash(key)];for(int i=0;i<b.size();i++)if(Objects.equals(b.get(i).key,key))return true;return false;}
    public V remove(K key){Objects.requireNonNull(key,"Key required");LinkedList<Entry<K,V>> b=buckets[hash(key)];for(int i=0;i<b.size();i++){Entry<K,V> e=b.get(i);if(Objects.equals(e.key,key)){b.removeAt(i);size--;return e.value;}}return null;}
    @SuppressWarnings("unchecked") public void resize(){capacity*=2;LinkedList<Entry<K,V>>[] old=buckets;buckets=new LinkedList[capacity];for(int i=0;i<capacity;i++)buckets[i]=new LinkedList<>();size=0;for(LinkedList<Entry<K,V>> b:old)for(int i=0;i<b.size();i++){Entry<K,V> e=b.get(i);put(e.key,e.value);}}
    public int size(){return size;}
}
```
