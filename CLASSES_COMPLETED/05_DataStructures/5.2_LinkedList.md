## Complexity
- Add first: O(1)
- Add last with tail: O(1)
- Search/access by index: O(n)
- Remove first: O(1)
- Remove at index: O(n)

import java.util.NoSuchElementException;
import java.util.Objects;
public class LinkedList<T> {
    private LinkedListNode<T> head,tail; private int size;
    public LinkedList(){} public void addFirst(T data){Objects.requireNonNull(data);LinkedListNode<T> n=new LinkedListNode<>(data);if(head==null)tail=n;else n.setNext(head);head=n;size++;}
    public void addLast(T data){Objects.requireNonNull(data);LinkedListNode<T> n=new LinkedListNode<>(data);if(tail==null)head=n;else tail.setNext(n);tail=n;size++;}
    public void insertAt(int index,T data){Objects.requireNonNull(data);if(index<0||index>size)throw new IndexOutOfBoundsException(index);if(index==0){addFirst(data);return;}if(index==size){addLast(data);return;}LinkedListNode<T> cur=head;for(int i=0;i<index-1;i++)cur=cur.getNext();LinkedListNode<T> n=new LinkedListNode<>(data);n.setNext(cur.getNext());cur.setNext(n);size++;}
    public T removeFirst(){if(head==null)throw new NoSuchElementException();T v=head.getData();head=head.getNext();if(head==null)tail=null;size--;return v;} public T removeLast(){if(tail==null)throw new NoSuchElementException();if(head==tail){T v=head.getData();clear();return v;}LinkedListNode<T> cur=head;while(cur.getNext()!=tail)cur=cur.getNext();T v=tail.getData();cur.setNext(null);tail=cur;size--;return v;}
    public T removeAt(int index){if(index<0||index>=size)throw new IndexOutOfBoundsException(index);if(index==0)return removeFirst();LinkedListNode<T> cur=head;for(int i=0;i<index-1;i++)cur=cur.getNext();LinkedListNode<T> doomed=cur.getNext();cur.setNext(doomed.getNext());if(doomed==tail)tail=cur;size--;return doomed.getData();}
    public T get(int index){if(index<0||index>=size)throw new IndexOutOfBoundsException(index);LinkedListNode<T> cur=head;for(int i=0;i<index;i++)cur=cur.getNext();return cur.getData();}
    public T getFirst(){if(head==null)throw new NoSuchElementException();return head.getData();} public T getLast(){if(tail==null)throw new NoSuchElementException();return tail.getData();}
    public boolean contains(T data){return indexOf(data)>=0;} public int indexOf(T data){LinkedListNode<T> cur=head;int i=0;while(cur!=null){if(Objects.equals(cur.getData(),data))return i;i++;cur=cur.getNext();}return -1;} public int size(){return size;} public boolean isEmpty(){return size==0;} public void clear(){head=tail=null;size=0;} 
}
