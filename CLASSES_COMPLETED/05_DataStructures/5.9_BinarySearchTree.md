Average operations are O(log n) for a reasonably balanced tree; worst case is O(n). A self-balancing tree is not implied by this class.

```java
import java.util.Comparator;
import java.util.Objects;
public class BinarySearchTree<T> {
 private TreeNode<T> root; private Comparator<T> comparator; public BinarySearchTree(Comparator<T> comparator){if(comparator==null)throw new IllegalArgumentException("Comparator required");this.comparator=comparator;}
 public void insert(T data){root=insert(root,Objects.requireNonNull(data));} private TreeNode<T> insert(TreeNode<T> n,T d){if(n==null)return new TreeNode<>(d);int c=comparator.compare(d,n.data);if(c<0)n.left=insert(n.left,d);else if(c>0)n.right=insert(n.right,d);return n;}
 public boolean search(T data){if(data==null)return false;TreeNode<T> n=root;while(n!=null){int c=comparator.compare(data,n.data);if(c==0)return true;n=c<0?n.left:n.right;}return false;}

}
```
