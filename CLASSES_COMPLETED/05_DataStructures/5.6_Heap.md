Define whether this is a min-heap or max-heap through the comparator. Heap insertion/removal are O(log n); peek is O(1).

```java
import java.util.Comparator;
import java.util.NoSuchElementException;
import java.util.Objects;
public class Heap<T> {
    private Object[] array; private int size; private Comparator<T> comparator;
    @SuppressWarnings("unchecked") public Heap(int capacity,Comparator<T> comparator){if(capacity<1||comparator==null)throw new IllegalArgumentException("Invalid heap");array=new Object[capacity];this.comparator=comparator;}
    public void insert(T value){Objects.requireNonNull(value,"Value required");if(size==array.length)grow();array[size++]=value;for(int i=size/2-1;i>=0;i--)heapify(i);}
    @SuppressWarnings("unchecked") public T extractRoot(){if(isEmpty())throw new NoSuchElementException("Heap is empty");T root=(T)array[0];array[0]=array[--size];array[size]=null;if(size>0)heapify(0);return root;}
    @SuppressWarnings("unchecked") public T peek(){if(isEmpty())throw new NoSuchElementException("Heap is empty");return (T)array[0];}
    @SuppressWarnings("unchecked") public void heapify(int index){int best=index,left=2*index+1,right=2*index+2;if(left<size&&comparator.compare((T)array[left],(T)array[best])<0)best=left;if(right<size&&comparator.compare((T)array[right],(T)array[best])<0)best=right;if(best!=index){Object t=array[index];array[index]=array[best];array[best]=t;heapify(best);}}
    public void buildHeap(){for(int i=size/2-1;i>=0;i--)heapify(i);} public void increasePriority(int index,T value){update(index,value);} public void decreasePriority(int index,T value){update(index,value);} 
    @SuppressWarnings("unchecked") private void update(int index,T value){if(index<0||index>=size||value==null)throw new IllegalArgumentException("Invalid heap update");array[index]=value;buildHeap();}
    private void grow(){Object[] n=new Object[array.length*2];System.arraycopy(array,0,n,0,array.length);array=n;} public boolean isEmpty(){return size==0;} public int size(){return size;}
}
```
