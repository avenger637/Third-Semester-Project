# 6.3 SchedulingAlgorithms.java



Conflict rule for half-open time intervals: two activities overlap if `start1 < end2 && start2 < end1`. Validate start before end.

```java
import java.time.LocalDateTime;
import java.util.Comparator;
public class SchedulingAlgorithms {
 public static void sortSchedule(LinkedList<ScheduleItem> schedule){if(schedule==null)return;ScheduleItem[] a=new ScheduleItem[schedule.size()];for(int i=0;i<a.length;i++)a[i]=schedule.get(i);SortingAlgorithms.mergeSort(a,Comparator.comparing(ScheduleItem::getStartTime));for(int i=0;i<a.length;i++){ScheduleItem v=a[i];for(int j=i-1;j>=0;j--){} } /* rebuild list */ while(!schedule.isEmpty())schedule.removeFirst();for(ScheduleItem item:a)schedule.addLast(item);}
 public static boolean detectConflict(ScheduleItem first,ScheduleItem second){if(first==null||second==null)return false;LocalDateTime a=first.getStartTime(),b=first.getEndTime(),c=second.getStartTime(),d=second.getEndTime();if(a==null||b==null||c==null||d==null||!b.isAfter(a)||!d.isAfter(c))throw new IllegalArgumentException("Invalid schedule interval");return a.isBefore(d)&&c.isBefore(b);}
 public static boolean detectAnyConflict(LinkedList<ScheduleItem> schedule){for(int i=0;i<schedule.size();i++)for(int j=i+1;j<schedule.size();j++)if(detectConflict(schedule.get(i),schedule.get(j)))return true;return false;}
 public static LinkedList<ScheduleItem> getSortedSchedule(LinkedList<ScheduleItem> schedule){LinkedList<ScheduleItem> r=new LinkedList<>();if(schedule!=null)for(int i=0;i<schedule.size();i++)r.addLast(schedule.get(i));sortSchedule(r);return r;}
 public static ScheduleItem getNextActivity(LinkedList<ScheduleItem> schedule,LocalDateTime currentTime){if(schedule==null||currentTime==null)return null;ScheduleItem next=null;for(int i=0;i<schedule.size();i++){ScheduleItem x=schedule.get(i);if(x.getStartTime()!=null&&x.getStartTime().isBefore(x.getEndTime())&&(!x.getStartTime().isBefore(currentTime)||x.getEndTime().isAfter(currentTime))&&(next==null||x.getStartTime().isBefore(next.getStartTime())))next=x;}return next;}
}
```
