public class RegistrationAlgorithms {
 public static boolean isAlreadyRegistered(LinkedList<Registration> registrations,int participantId,int eventId){if(registrations==null)return false;for(int i=0;i<registrations.size();i++){Registration r=registrations.get(i);if(r.getParticipantId()==participantId&&r.getEventId()==eventId&&r.getStatus()==RegistrationStatus.REGISTERED)return true;}return false;}
 public static int countActiveRegistrations(LinkedList<Registration> registrations){int n=0;if(registrations!=null)for(int i=0;i<registrations.size();i++)if(registrations.get(i).getStatus()==RegistrationStatus.REGISTERED)n++;return n;} public static int countCancelledRegistrations(LinkedList<Registration> registrations){int n=0;if(registrations!=null)for(int i=0;i<registrations.size();i++)if(registrations.get(i).getStatus()==RegistrationStatus.CANCELLED)n++;return n;}
 public static LinkedList<Registration> filterActive(LinkedList<Registration> registrations){LinkedList<Registration> r=new LinkedList<>();if(registrations!=null)for(int i=0;i<registrations.size();i++)if(registrations.get(i).getStatus()==RegistrationStatus.REGISTERED)r.addLast(registrations.get(i));return r;} public static LinkedList<Registration> filterCancelled(LinkedList<Registration> registrations){LinkedList<Registration> r=new LinkedList<>();if(registrations!=null)for(int i=0;i<registrations.size();i++)if(registrations.get(i).getStatus()==RegistrationStatus.CANCELLED)r.addLast(registrations.get(i));return r;}
}
