# 6.1 SearchingAlgorithms.java



Linear search: O(n). Binary search: O(log n), but requires sorted input.

```java
import java.util.Comparator;
import java.util.function.ToIntFunction;
import java.util.function.Function;
public class SearchingAlgorithms {
 public static <T> int linearSearch(T[] array,T target,Comparator<T> comparator){if(array==null||target==null||comparator==null)return -1;for(int i=0;i<array.length;i++)if(comparator.compare(array[i],target)==0)return i;return -1;}
 public static <T> int binarySearch(T[] array,T target,Comparator<T> comparator){if(array==null||target==null||comparator==null)return -1;int lo=0,hi=array.length-1;while(lo<=hi){int m=lo+(hi-lo)/2,c=comparator.compare(array[m],target);if(c==0)return m;if(c<0)lo=m+1;else hi=m-1;}return -1;}
 public static <T> T searchById(LinkedList<T> list,int id,ToIntFunction<T> idExtractor){for(int i=0;i<list.size();i++){T v=list.get(i);if(idExtractor.applyAsInt(v)==id)return v;}return null;}
 public static <T> T searchByName(LinkedList<T> list,String name,Function<T,String> nameExtractor){if(name==null)return null;for(int i=0;i<list.size();i++){T v=list.get(i);if(name.equalsIgnoreCase(nameExtractor.apply(v)))return v;}return null;}
}
```
