# 6.2 SortingAlgorithms.java



Suggested use: sort events by date, participants by name, and expenses by amount. Merge sort is O(n log n); quicksort averages O(n log n) but can be O(n²) in the worst case.

```java
import java.util.Comparator;
public class SortingAlgorithms {
 public static <T> void mergeSort(T[] array,Comparator<T> comparator){if(array==null||comparator==null)return;T[] tmp=array.clone();mergeSort(array,tmp,0,array.length,comparator);} private static <T> void mergeSort(T[] a,T[] tmp,int lo,int hi,Comparator<T> c){if(hi-lo<2)return;int m=(lo+hi)/2;mergeSort(a,tmp,lo,m,c);mergeSort(a,tmp,m,hi,c);int i=lo,j=m,k=lo;while(i<m&&j<hi)tmp[k++]=c.compare(a[i],a[j])<=0?a[i++]:a[j++];while(i<m)tmp[k++]=a[i++];while(j<hi)tmp[k++]=a[j++];System.arraycopy(tmp,lo,a,lo,hi-lo);}
 public static <T> void quickSort(T[] a,Comparator<T> c){if(a!=null&&c!=null)quickSort(a,0,a.length-1,c);} private static <T> void quickSort(T[] a,int lo,int hi,Comparator<T> c){if(lo>=hi)return;int p=partition(a,lo,hi,c);quickSort(a,lo,p-1,c);quickSort(a,p+1,hi,c);} private static <T> int partition(T[] a,int lo,int hi,Comparator<T> c){T pivot=a[hi];int i=lo;for(int j=lo;j<hi;j++)if(c.compare(a[j],pivot)<=0){T t=a[i];a[i]=a[j];a[j]=t;i++;}T t=a[i];a[i]=a[hi];a[hi]=t;return i;}
 public static <T> void heapSort(T[] a,Comparator<T> c){if(a==null||c==null)return;for(int i=a.length/2-1;i>=0;i--)heapify(a,i,a.length,c);for(int i=a.length-1;i>0;i--){T t=a[0];a[0]=a[i];a[i]=t;heapify(a,0,i,c);}} private static <T> void heapify(T[] a,int i,int n,Comparator<T> c){int best=i,l=2*i+1,r=2*i+2;if(l<n&&c.compare(a[l],a[best])>0)best=l;if(r<n&&c.compare(a[r],a[best])>0)best=r;if(best!=i){T t=a[i];a[i]=a[best];a[best]=t;heapify(a,best,n,c);}}
 public static <T> void insertionSort(T[] a,Comparator<T> c){if(a==null||c==null)return;for(int i=1;i<a.length;i++){T v=a[i];int j=i-1;while(j>=0&&c.compare(a[j],v)>0){a[j+1]=a[j];j--;}a[j+1]=v;}}
 public static <T> void bubbleSort(T[] a,Comparator<T> c){if(a==null||c==null)return;for(int i=0;i<a.length;i++){boolean swapped=false;for(int j=1;j<a.length-i;j++)if(c.compare(a[j-1],a[j])>0){T t=a[j-1];a[j-1]=a[j];a[j]=t;swapped=true;}if(!swapped)return;}}
}
```
