# 6.4 BudgetAlgorithms.java



Use `BigDecimal`, reject negative values, and ensure budget checks and expense persistence occur consistently in a transaction.

```java
import java.math.*;
import java.util.Comparator;
public class BudgetAlgorithms {
 private static void requireMoney(BigDecimal v){if(v==null||v.signum()<0)throw new IllegalArgumentException("Money must be nonnegative");}
 public static BigDecimal calculateTotalExpenses(LinkedList<Expense> expenses){BigDecimal total=BigDecimal.ZERO;if(expenses!=null)for(int i=0;i<expenses.size();i++){Expense e=expenses.get(i);if(e!=null&&e.getAmount()>=0)total=total.add(BigDecimal.valueOf(e.getAmount()).setScale(2,RoundingMode.HALF_UP));}return total.setScale(2,RoundingMode.HALF_UP);}
 public static BigDecimal calculateRemainingBudget(BigDecimal allocated,BigDecimal spent){requireMoney(allocated);requireMoney(spent);return allocated.subtract(spent).setScale(2,RoundingMode.HALF_UP);}
 public static BigDecimal calculateUtilization(BigDecimal allocated,BigDecimal spent){requireMoney(allocated);requireMoney(spent);if(allocated.signum()==0)return BigDecimal.ZERO.setScale(2);return spent.divide(allocated,4,RoundingMode.HALF_UP).multiply(BigDecimal.valueOf(100)).setScale(2,RoundingMode.HALF_UP);}
 public static boolean checkBudgetLimit(BigDecimal allocated,BigDecimal spent,BigDecimal newExpense){try{requireMoney(allocated);requireMoney(spent);requireMoney(newExpense);return spent.add(newExpense).compareTo(allocated)<=0;}catch(IllegalArgumentException e){return false;}}
 public static LinkedList<Expense> sortExpenses(LinkedList<Expense> expenses){LinkedList<Expense> r=new LinkedList<>();if(expenses==null)return r;Expense[] a=new Expense[expenses.size()];for(int i=0;i<a.length;i++)a[i]=expenses.get(i);SortingAlgorithms.mergeSort(a,Comparator.comparing(e->BigDecimal.valueOf(e.getAmount())));for(Expense e:a)r.addLast(e);return r;}
 public static Expense findLargestExpense(LinkedList<Expense> expenses){if(expenses==null||expenses.isEmpty())return null;Expense r=expenses.get(0);for(int i=1;i<expenses.size();i++)if(expenses.get(i).getAmount()>r.getAmount())r=expenses.get(i);return r;}
}
```
