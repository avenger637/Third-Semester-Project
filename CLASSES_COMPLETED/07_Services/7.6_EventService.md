Keep cache/index consistency after create, update, or delete. Cache only data the current user's authorization permits.

```java
public class EventService { private final EventRepository eventRepository; private final LinkedList<Event> eventCache=new LinkedList<>(); private final HashTable<Integer,Event> eventIndex=new HashTable<>(16); public EventService(EventRepository r){eventRepository=java.util.Objects.requireNonNull(r);refreshEventCache();} public Event createEvent(Event e){if(e==null)throw new ValidationException("Event required");eventRepository.save(e);eventCache.addLast(e);eventIndex.put(e.getEventId(),e);return e;} public Event getEventById(int id){return eventIndex.get(id);} public LinkedList<Event> getAllEvents(){return eventCache;} public LinkedList<Event> getEventsByOrganizer(int id){return eventRepository.findByOrganizer(id);} public LinkedList<Event> searchEventsByName(String n){LinkedList<Event> r=new LinkedList<>();for(int i=0;i<eventCache.size();i++)if(eventCache.get(i).getTitle().toLowerCase().contains(n.toLowerCase()))r.addLast(eventCache.get(i));return r;} public LinkedList<Event> sortEventsByDate(){return eventCache;} public boolean updateEvent(Event e){if(e==null)return false;boolean ok=eventRepository.update(e);if(ok){eventIndex.put(e.getEventId(),e);refreshEventCache();}return ok;} public boolean deleteEvent(int id){boolean ok=eventRepository.delete(id);if(ok){eventIndex.remove(id);refreshEventCache();}return ok;} public void refreshEventCache(){eventCache.clear();LinkedList<Event> all=eventRepository.findAll();for(int i=0;i<all.size();i++){eventCache.addLast(all.get(i));eventIndex.put(all.get(i).getEventId(),all.get(i));}} public void rebuildEventIndex(){for(int i=0;i<eventCache.size();i++)eventIndex.put(eventCache.get(i).getEventId(),eventCache.get(i));} }
```
