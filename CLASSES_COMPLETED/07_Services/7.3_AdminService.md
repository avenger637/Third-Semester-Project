Every administrative operation must require an Admin role.

```java
public class AdminService { public AdminService(Object... dependencies){} public Event createEvent(Event e){return e;} public boolean assignOrganizer(int eventId,int organizerId){return eventId>0&&organizerId>0;} public boolean allocateBudget(int eventId,java.math.BigDecimal amount){return eventId>0&&amount!=null&&amount.signum()>=0;} public LinkedList<Event> getAllEvents(){return new LinkedList<>();} public LinkedList<Organizer> getAllOrganizers(){return new LinkedList<>();} public java.math.BigDecimal getOverallAllocatedBudget(){return java.math.BigDecimal.ZERO;} public java.math.BigDecimal getOverallExpenses(){return java.math.BigDecimal.ZERO;} public java.math.BigDecimal getOverallIncome(){return java.math.BigDecimal.ZERO;} public java.math.BigDecimal getOverallBalance(){return java.math.BigDecimal.ZERO;} public boolean payOrganizer(int id,java.math.BigDecimal amount){return id>0&&amount!=null&&amount.signum()>=0;} }
```
