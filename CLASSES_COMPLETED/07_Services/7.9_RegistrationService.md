Use a consistent composite key (participant ID + event ID). The database must also enforce duplicate-registration integrity.

```java
public class RegistrationService { private final RegistrationRepository repository; private final HashTable<String,Registration> registrationIndex=new HashTable<>(16); public RegistrationService(RegistrationRepository r){repository=r;rebuildRegistrationIndex();} private String key(int p,int e){return p+":"+e;} public Registration register(int p,int e){if(isRegistered(p,e))throw new RegistrationException("Already registered");Registration r=new Registration(0,p,e);repository.save(r);registrationIndex.put(key(p,e),r);return r;} public boolean cancel(int p,int e){Registration r=registrationIndex.get(key(p,e));if(r==null)return false;r.cancel();return repository.update(r);} public boolean isRegistered(int p,int e){Registration r=registrationIndex.get(key(p,e));return r!=null&&r.getStatus()==RegistrationStatus.ACTIVE;} public LinkedList<Registration> getByEvent(int e){return repository.findByEvent(e);} public LinkedList<Registration> getByParticipant(int p){return repository.findByParticipant(p);} public LinkedList<Registration> getActiveRegistrations(int e){return RegistrationAlgorithms.filterActive(getByEvent(e));} public LinkedList<Registration> getCancelledRegistrations(int e){return RegistrationAlgorithms.filterCancelled(getByEvent(e));} public int countParticipants(int e){return getActiveRegistrations(e).size();} public void rebuildRegistrationIndex(){LinkedList<Registration> all=repository.findAll();for(int i=0;i<all.size();i++)registrationIndex.put(key(all.get(i).getParticipantId(),all.get(i).getEventId()),all.get(i));} }
```
