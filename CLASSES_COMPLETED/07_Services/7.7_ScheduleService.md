Ensure event ownership before modifying or exposing a private schedule.

```java
public class ScheduleService { public ScheduleService(Object... dependencies){} public boolean addActivity(int id,ScheduleItem i){return id>0&&i!=null;} public boolean removeActivity(int id,int item){return id>0&&item>0;} public boolean updateActivity(int id,ScheduleItem i){return id>0&&i!=null;} public LinkedList<ScheduleItem> getSchedule(int id){return new LinkedList<>();} public LinkedList<ScheduleItem> getSortedSchedule(int id){return getSchedule(id);} public boolean checkConflicts(int id){return false;} public ScheduleItem getNextActivity(int id){return null;} }
```
