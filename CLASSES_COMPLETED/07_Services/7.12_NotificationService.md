Ensure notification reads/updates are scoped to the authenticated recipient.

```java
public class NotificationService { private final HashTable<Integer,LinkedList<Notification>> notificationIndex=new HashTable<>(16); public Notification notifyUser(int id,String message){Notification n=new Notification(0,id,message);LinkedList<Notification> list=notificationIndex.get(id);if(list==null){list=new LinkedList<>();notificationIndex.put(id,list);}list.addLast(n);return n;} public LinkedList<Notification> getNotifications(int id){LinkedList<Notification> n=notificationIndex.get(id);return n==null?new LinkedList<>():n;} public boolean markAsRead(int id){return false;} public void markAllAsRead(int id){LinkedList<Notification> n=getNotifications(id);for(int i=0;i<n.size();i++)n.get(i).markAsRead();} }
```
